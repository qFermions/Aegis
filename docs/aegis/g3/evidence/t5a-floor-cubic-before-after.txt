// The 12 guard floors and 10 engine twins that chained two gaps, each timed on a line

// on 20 KB (before on 20 KB takes 11 to 44 s per two-gap floor and is skipped here).
const fs = require('fs');
const S = process.argv[2];
const load = (file) => [...fs.readFileSync(file, 'utf8').matchAll(/^\s*\{ pattern: (\/.*\/[a-z]*), label: '((?:[^'\\]|\\.)*)'/gm)].map((m) => ({ re: eval(m[1]), label: m[2] }));
const cases = [['group membership removal', 'remove member from '], ['security-control change (SR-2)', 'exclude from '], ['account / work-profile removal', 'remove account from '],
  ['security-control change -- Google', 'exclude from '], ['Slack or Linear member', 'kick user '], ['Slack or Linear member', 'slack kick '], ['Okta lifecycle change', 'reset all '],

  ['fleet-wide computer or Mac', 'lock all ']];
for (const [name, file] of [['guard', 'scripts/tier/tier-guard.js'], ['engine', 'scripts/graph/engine.js']]) {
  const before = load(`${S}/head/${file.split('/').pop()}`), after = load(file);
  for (const [prefix, unit] of cases) {
    const i = before.findIndex((p) => p.label.startsWith(prefix));
    if (i < 0 || before[i].re.source === after[i].re.source) continue;
    const line = (n) => unit.repeat(Math.ceil(n / unit.length)).slice(0, n);
    const ms = (re, n) => { const s = line(n); const t = process.hrtime.bigint(); re.test(s); return (Number(process.hrtime.bigint() - t) / 1e6).toFixed(0); };
    console.log(`${name.padEnd(6)} ${prefix.padEnd(32)} ${JSON.stringify(unit).padEnd(22)} 5 KB before ${ms(before[i].re, 5000).padStart(5)} ms, after ${ms(after[i].re, 5000).padStart(3)} ms; 20 KB after ${ms(after[i].re, 20000).padStart(3)} ms`);
  }
}
