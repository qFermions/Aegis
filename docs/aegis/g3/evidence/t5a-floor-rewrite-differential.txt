// gapCount(source): how many unbounded gaps ([^.]*, .*, [^\r\n]*) one match can chain in
// sequence: a group counts its widest alternative, a lookaround counts 0 here and is
// checked on its own. problems(source) lists every place the count reaches 2.
function parseGroup(s, i) { // s[i] is just after '(' or at start; returns { alts: [[items]], end }
  const alts = [[]];
  while (i < s.length && s[i] !== ')') {
    const c = s[i];
    if (c === '|') { alts.push([]); i++; continue; }
    if (c === '\\') { alts[alts.length - 1].push({ t: 'atom' }); i += 2; continue; }
    if (c === '[') { let j = i + 1; if (s[j] === '^') j++; if (s[j] === ']') j++; while (s[j] !== ']') { if (s[j] === '\\') j++; j++; } const cls = s.slice(i, j + 1); i = j + 1; alts[alts.length - 1].push(quant(s, i, { t: cls === '[^.]' || cls === '[^\\r\\n]' ? 'gapclass' : 'atom' })); i = alts[alts.length - 1].at(-1).next; continue; }
    if (c === '(') {
      const look = /^\(\?<?[=!]/.test(s.slice(i, i + 4));
      const start = i + (look ? (s[i + 2] === '<' ? 4 : 3) : (s[i + 1] === '?' ? (s[i + 2] === '<' ? s.indexOf('>', i) + 1 - i : 3) : 1));
      const g = parseGroup(s, start);
      i = g.end + 1;
      alts[alts.length - 1].push(quant(s, i, { t: look ? 'look' : 'group', alts: g.alts }));
      i = alts[alts.length - 1].at(-1).next; continue;
    }
    alts[alts.length - 1].push(quant(s, i + 1, { t: c === '.' ? 'gapclass' : 'atom' }));
    i = alts[alts.length - 1].at(-1).next;
  }
  return { alts, end: i };
}
function quant(s, i, item) {
  let q = '';
  if (s[i] === '*' || s[i] === '+' || s[i] === '?') { q = s[i]; i++; } else if (s[i] === '{') { const j = s.indexOf('}', i); q = s.slice(i, j + 1); i = j + 1; }
  if (s[i] === '?') i++;
  return { ...item, q, unbounded: q === '*' || q === '+' || /,\}$/.test(q), next: i };
}
function count(items, problems, where) {
  let n = 0;
  for (const it of items) {
    if (it.t === 'gapclass' && it.unbounded) n += 1;
    else if (it.t === 'look') for (const a of it.alts) { if (count(a, problems, 'lookaround') > 1) problems.push(where + ' > lookaround'); }
    else if (it.t === 'group') { const inner = Math.max(...it.alts.map((a) => count(a, problems, where))); n += it.unbounded && inner > 0 ? 2 : inner; }
  }
  return n;
}
function problems(source) {

  g.alts.forEach((a, k) => { if (count(a, out, `alternative ${k + 1}`) > 1) out.push(`alternative ${k + 1}`); });
  return out;
}
